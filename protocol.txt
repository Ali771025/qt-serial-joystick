reserve[0]=0-> mode1
reserve[0]=1-> mode2
data_val=12 -> mode1 TX
data_val=3 -> mode2 TX
data_val -> ignored for RX interpretation (still included in CRC)
V_sph.X	-> mode1 x
V_sph.Y	-> mode1 y
pos_sph.X -> mode2 x
pos_sph.Y -> mode2 y
reserve[1]=1 -> homming
reserve[2]=1 -> trig
reserve[3]=0..255 -> UI value
reserve[4]=0 -> reserved
CRC = XOR of data[1] through data[30]; data[31]=CRC; data[32]=0x5A
